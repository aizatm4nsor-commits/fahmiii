    <script>
        const ctx = document.getElementById('growthChart').getContext('2d');
        const labels = Array.from({length: 16}, (_, i) => i + 5);
        const dataPoints = labels.map(x => 0.05 * Math.pow(x, 3) - 1.2 * Math.pow(x, 2) + 10 * x + 2);

        const chart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'FFB Mass (kg)',
                    data: dataPoints,
                    borderColor: '#047857',
                    backgroundColor: '#059669',
                    tension: 0.3,
                    pointRadius: 6
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                onClick: (event, elements) => {
                    if (elements.length > 0) {
                        const index = elements[0].index;
                        const x = labels[index];
                        const y = dataPoints[index].toFixed(2);
                        
                        const derivative = (0.15 * Math.pow(x, 2) - 2.4 * x + 10).toFixed(2);
                        
                        const panel = document.getElementById('resultsPanel');
                        const text = document.getElementById('analysisText');
                        
                        panel.classList.remove('hidden');
                        text.innerHTML = `At Week <b>${x}</b>, the total weight is <b>${y} kg</b>. <br/>
                                         The instantaneous growth rate is <b>${derivative} kg/week</b>.<br/>
                                         <span class="text-sm opacity-80 mt-2 block">${derivative > 0 ? 'The bunch is still actively gaining mass.' : 'The growth rate has stabilized or started to decline; harvest might be imminent.'}</span>`;
                    }
                }
            }
        });
    </script>
